crud_qr_project/          # Carpeta raíz del proyecto
│── manage.py             # Script para administrar el proyecto Django

│   │── settings.py       # Configuración de Django (INSTALLED_APPS, DATABASES, etc.)
│   │── urls.py           # Enrutamiento global del proyecto
│   │── wsgi.py           # Archivo WSGI para servidores de producción
│   │── asgi.py           # Archivo ASGI para WebSockets u otras aplicaciones asíncronas
│── app/                  # Aplicación principal donde está el CRUD
│   │── __init__.py
│   │── models.py         # Definición de los modelos (User, Item con QR)
│   │── views.py          # Lógica de las vistas con DRF
│   │── serializers.py    # Serializadores para transformar datos JSON
│   │── urls.py           # Enrutamiento de la API
│   │── permissions.py    # Permisos personalizados para roles de usuario
│   │── admin.py          # Configuración de modelos en el panel de administración
│   │── tests.py          # Pruebas unitarias
│   │── apps.py           # Configuración de la app Django
│── static/               # Archivos estáticos (CSS, JS, imágenes)
│── media/                # Archivos subidos (QR generados)
│── env/                  # (Opcional) Entorno virtual de Python
